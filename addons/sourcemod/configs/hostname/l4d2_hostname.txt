"hostname"
{
	// 按"IP:端口"分服: 同一份文件可以给多台服务器用, 每台服只取自己那一段.
	// 查找顺序: "IP:端口" -> status里本地绑定地址的"IP:端口" -> "IP:*" -> "*:端口" -> "default"
	//
	// IP和端口的识别顺序(前3步都不执行任何控制台指令):
	//   1. l4d2_hostname_ip 变量(手动指定, 应急用)
	//   2. 引擎签名直读 NET_GetPublicAdr  (变量 l4d2_hostname_engine, 默认1=开启;
	//      签名在 addons/sourcemod/gamedata/l4d2_hostname.txt, 取到的就是status里public那段)
	//   3. net_public_adr 变量
	//   4. 上面都失败才执行 status 指令, 从
	//      "udp/ip  : x.x.x.x:端口 [ public x.x.x.x:端口 ]" 里收集(公网优先, 其次本地绑定)
	//   5. 最后才用 ip / hostip 这两个绑定地址兜底
	//
	// 不知道当前服务器用哪个键: 游戏里输入 !host, 提示里会显示"当前配置键";
	// 也可以看 addons/sourcemod/logs/ 下当天的 SourceMod 日志(搜索"当前配置键").

	"default"
	{
		"ServerName"		"苹果果的抽象进阶包抗服"
	}

	// ===== 完整写法: 指定 IP + 指定端口 (最精确, 优先级最高) =====
	"1.15.122.209:27015"
	{
		"ServerName"		"抽象进阶包抗服1服"
	}

	"1.15.62.95:27015"
	{
		"ServerName"		"抽象进阶包抗服2服"
	}

	"61.147.247.7:27016"
	{
		"ServerName"		"抽象进阶包抗服3服"
	}

	"180.188.21.24:27030"
	{
		"ServerName"		"抽象进阶包抗服4服"
	}

	"180.188.21.24:27031"
	{
		"ServerName"		"抽象进阶包抗服5服"
	}

	"180.188.21.24:27033"
	{
		"ServerName"		"抽象进阶包抗服6服"
	}

	// ===== 通配写法 (不写具体某一台, 兜底用; 优先级低于完整写法) =====
	//
	// 1) 完整IP + 通配端口 "IP:*": 这台机器/IP 上的所有端口共用一个服名.
	//    适合同一台机器开了很多端口、又懒得一条条写的情况;
	//    也可以只给它配一段, 比如 4服/5服 同在 180.188.21.24, 想统一叫"多服X号"时用.
	// "180.188.21.24:*"
	// {
	//     "ServerName"	"抽象进阶包抗服(21.24端口合集)"
	// }
	//
	// 2) 通配IP + 指定端口 "*:端口": 所有机器上跑这个端口的服务器共用一个服名.
	// "*:27015"
	// {
	//     "ServerName"	"27015端口服"
	// }
	//
	// 提示: 通配段只影响"没有匹配到更精确键"的服务器; 完整写法永远优先.
	//       想确认某台服实际用了哪个键, 在它上面输入 !host 看"当前配置键".
}
