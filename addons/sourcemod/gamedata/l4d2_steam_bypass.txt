		"Offsets"
		{
			/*
			 * How this plugin reaches the IClient virtuals (GetNetChannel, Disconnect) differs per
			 * platform, because the two builds are laid out differently:
			 *   windows - IClient is the subobject at +4 (RTTI base class mdisp = 4) and GetNetChannel
			 *             is slot 18 of that frame (CBaseServer::CheckTimeouts, engine.dll RVA 0x135880,
			 *             calls (*(client+4))[18]).
			 *   linux   - the plugin uses the primary CBaseClient vtable instead, where GetNetChannel is
			 *             slot 8 (_ZTV11CBaseClient, verified) and Disconnect resolves through its own
			 *             symbol, so no adjustment is applied.
			 * This is therefore this plugin's frame offset, not the engine's mdisp.
			 */
			"CBaseClient->IClient->frame_offset"
			{
				"linux"		"0"
				"windows"	"4"
			}

			/*
			 * The engine's "fully authenticated" byte, +0x217 on BOTH platforms. The success branch
			 * of the auth callback writes 1 into it and the failure branch jumps past that write:
			 *   linux   0x20137C  mov byte ptr [ebx+217h], 1   (failure path jumps to 0x2013F0)
			 *   windows 0x12E78E  mov byte ptr [ebx+217h], 1
			 * and it is exactly what SourceMod's auth gate reads:
			 *   CVEngineServer::IsClientFullyAuthenticated -> movzx eax, byte ptr [edx+217h]
			 * So code 0 => flag 1 => SourceMod authorises the client and fires OnClientAuthorized /
			 * OnClientPostAdminCheck. The plugin logs the flag around its rewrite, which turns that
			 * log line into an end-to-end proof.
			 */
			"CBaseClient->m_bFullyAuthenticated"
			{
				"linux"		"535"
				"windows"	"535"
			}

			/*
			 * ValidateAuthTicketResponse_t { CSteamID m_SteamID; EAuthSessionResponse m_eAuthSessionResponse; CSteamID m_OwnerSteamID; }
			 * id (8 bytes) is at +0, the verdict is at +8. Both platforms verified:
			 *   windows engine.dll RVA 0x12E622   mov esi, [esi+8]                       ; then jz <success path>
			 *   linux  engine_srv.so  RVA 0x201301  mov eax, [esi+8] ; test ; jnz <failure path>
			 * This engine build never reads m_OwnerSteamID (checked every [esi+..] access in the callback).
			 */
			"ValidateAuthTicketResponse_t->m_eAuthSessionResponse"
			{
				"linux"		"8"
				"windows"	"8"
			}

			/*
			 * CBaseClient::m_SteamID - 8 bytes: low dword first, then the high dword.
			 *   windows: CBaseClient::GetSteamID = IClient frame slot 16 (engine.dll RVA 0x4E2E0)
			 *            mov edx, [ecx+79h]        ; ecx is the IClient subobject = complete object + 4
			 *   linux:   CBaseClient::SetSteamID (engine_srv.so RVA 0xEA0B0)
			 *            mov [ecx+81h], edx / mov [ecx+7Dh], eax   ; ecx is the complete object
			 * => 0x7D from the complete object on BOTH platforms (this is also what l4dtoolz's
			 *    steam_off 0x7D uses; the engine itself reaches it as IClient+0x79).
			 */
			"CBaseClient->m_SteamID"
			{
				"linux"		"125"
				"windows"	"125"
			}

			/*
			 * char m_Name[32] inside CBaseClient, read straight from the complete object pointer.
			 * windows verified: the same buffer a name getter returns (engine.dll RVA 0x4FCD0: lea eax,[ecx+38h]).
			 */
			"CBaseClient->m_Name"
			{
				"linux"		"60"
				"windows"	"60"

			/*
			 * CNetChan::IsTimingOut - virtual index 7. Called on the CNetChan* that GetNetChannel

			 * windows verified (engine.dll RVA 0x1B1D50): last != -1.0 && now > last + 4.0

			"CNetChan::IsTimingOut"

					}

					{
						"type"	"int"
					}
